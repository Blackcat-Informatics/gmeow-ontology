gmeow:cueToneWorried
    a gmeow:Cue ;
    rdfs:isDefinedBy <https://blackcatinformatics.ca/gmeow/slices/lillith-observations> ;
    rdfs:label "worried"@en ;
    gmeow:description "Evidence that the speaker is troubled: the words say the speaker is worried or anxious. The refiner reads a counted match as troubled words, a reading and never how the speaker is."@en ;
    gmeow:detectionGuidance "Find \"worried\", \"worrying\" or \"anxious\" where the speaker says it of themself now (\"I'm worried about the roof\", \"feeling anxious tonight\"). It is not the cue when it is said of someone else, denied (\"not worried at all\"), in the past and over (\"I was worried, but it's fine now\"), or reported or quoted speech. Report as missed any words of the speaker's present worry using these words that were not reported."@en ;
    gmeow:cuePhrase "worried"@en , "worrying"@en , "anxious"@en ;
    gmeow:cueEvidenceFor gmeow:toneTroubled ;
    gmeow:contentLanguageCode "en"^^xsd:token ;
    gmeow:cueExample "I'm worried about the roof leaking again tonight."@en ;
    gmeow:cueCounterExample "The roof held up fine in the storm."@en ;
    gmeow:graphBoxRole gmeow:boxABox .
